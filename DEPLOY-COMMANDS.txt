# Paste this:

#!/bin/bash
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR=~/backups
mkdir -p $BACKUP_DIR

docker exec postgres_prod pg_dump -U quidpath_user quidpath_db > $BACKUP_DIR/backend_$DATE.sql
docker exec postgres_billing_prod pg_dump -U billing_user billing_prod > $BACKUP_DIR/billing_$DATE.sql
docker exec tazama_postgres pg_dump -U tazama_user tazama_db > $BACKUP_DIR/tazama_$DATE.sql

find $BACKUP_DIR -name "*.sql" -mtime +7 -delete

# Save and make executable

# Add to crontab
crontab -e
